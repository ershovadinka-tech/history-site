            ["2008 г.", "Барак Обама."]
          ],
          note: "Один из главных центров мировой политики."
        }
      }
    };

    const regionButtons = document.querySelectorAll('.region-btn');
    const periodButtons = document.querySelectorAll('.period-btn');
    const contentArea = document.getElementById('content-area');

    let currentRegion = 'world';
    let currentPeriod = 'ancient';

    function render() {
      regionButtons.forEach(b => b.classList.toggle('active', b.dataset.region === currentRegion));
      periodButtons.forEach(b => b.classList.toggle('active', b.dataset.period === currentPeriod));

      const block = data[currentRegion] && data[currentRegion][currentPeriod];

      if (!block) {
        contentArea.innerHTML = '<p class="empty">Для этого региона и периода пока нет данных.</p>';
        return;
      }

      const eventsHtml = block.events.map(([date, desc]) => `
        <div class="event-row">
          <span class="event-date">${date}</span>
          <p class="event-desc">${desc}</p>
        </div>
      `).join('');

      contentArea.innerHTML = `
        <div class="period-card visible">
          <div class="period-title">${block.title}</div>
          ${eventsHtml}
          <p class="note">${block.note}</p>
        </div>
      `;
    }

    regionButtons.forEach(btn => {
      btn.addEventListener('click', () => {
        currentRegion = btn.dataset.region;
        render();
      });
    });

    periodButtons.forEach(btn => {
      btn.addEventListener('click', () => {
        currentPeriod = btn.dataset.period;
        render();
      });
    });

    render();
  </script>
